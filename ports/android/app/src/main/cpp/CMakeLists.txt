cmake_minimum_required(VERSION 3.22.1)
project(sprechklart LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
if(NOT EXISTS "${WHISPER_SOURCE_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "Pinned whisper.cpp source missing. See ports/README.md.")
endif()
add_subdirectory("${WHISPER_SOURCE_DIR}" whisper-build)
add_library(sprechklart SHARED native.cpp)
target_link_libraries(sprechklart PRIVATE whisper android log)
target_link_options(sprechklart PRIVATE "-Wl,-z,max-page-size=16384")
